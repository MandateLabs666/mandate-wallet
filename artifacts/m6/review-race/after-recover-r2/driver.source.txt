  server=await serveWorkspace(workspace,{staticDirectory:web,authenticate:t=>{if(t!==token)throw Error("TEST_AUTH_REQUIRED");}});
  browser=await chromium.launch({channel:"chromium",headless:true});const page=await browser.newPage({viewport:{width:1280,height:900}});

    if(tx.from.toLowerCase()!==e.owner.address.toLowerCase()||tx.to.toLowerCase()!==e.wallet.toLowerCase()||tx.chainId!=="0x7a69"||tx.value!=="0x0")throw Error("LOCAL_FIXTURE_SCOPE_MISMATCH");

  });
  await page.addInitScript('window.ethereum={request:(args)=>window.controlledOwnerRequest(args)};');

  const nonceBefore=await e.pc.getTransactionCount({address:e.owner.address}),cashBefore=await e.pc.readContract({address:e.asset,abi:testTokenAbi,functionName:"balanceOf",args:[e.wallet]});

  await page.locator("#close-review").click();let b:any,heldResponse:any,releaseResponse:(()=>void)|undefined;

    b=await uiOperation("owner-action",'[data-action="revokeSession"]');await page.locator("#review").waitFor();
  }else if(scenario==="task"){
    await page.locator('[data-page="tasks"]').click();await uiOperation("snapshot",`[data-action="select"][data-id="${other.task.id}"]`);await page.locator("#content").filter({hasText:other.task.description}).waitFor();
  }else{

    if(await page.locator("#review").isVisible())await page.locator("#close-review").click();

    recoveryResult={newerReviewOpen:await page.locator("#review").isVisible(),reviewError:await page.locator("#review-error").innerText(),newerCallUnchanged:await page.locator("#review-call").textContent()===b.request.draft.transaction.data};
